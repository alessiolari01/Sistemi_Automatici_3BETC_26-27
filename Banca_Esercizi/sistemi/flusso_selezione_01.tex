% ---------------------------------------------------------------------
% Argomento: diagramma di flusso con calcolo e due decisioni annidate
%   fila A: forno, media di due temperature -> resistenza / ventola / ok
%   fila B: serbatoio, percentuale di riempimento -> carico / scarico / ok
%   fila C: totale di un acquisto con sconto del 20 % o del 10 %
% Livello: base (con suggerimenti nel testo)   |   Tempo stimato: 15 min
% ---------------------------------------------------------------------
\begin{esercizio}[6]
\ABC{%
In un forno due sensori misurano la temperatura. Disegna il diagramma di flusso di un algoritmo
che decide cosa fare in base alla temperatura media.\\
\textbf{Dati da leggere:} $T_1$, $T_2$ = temperature misurate dai due sensori (in °C).\\
\textbf{Altre variabili:} $T$ = media delle due temperature, cioè $T = (T_1+T_2)/2$.\\
Prima controlla se $T < 180$: in questo caso visualizza ``Accendi la resistenza''. Altrimenti
controlla se $T > 220$: in questo caso visualizza ``Accendi la ventola''; in caso contrario visualizza
``Temperatura corretta''.}{%
In un serbatoio un sensore misura il livello dell'acqua. Disegna il diagramma di flusso di un
algoritmo che decide cosa fare in base alla percentuale di riempimento.\\
\textbf{Dati da leggere:} $L$ = livello dell'acqua (in cm); $H$ = altezza del serbatoio (in cm).\\
\textbf{Altre variabili:} $P$ = percentuale di riempimento, cioè $P = L/H \cdot 100$.\\
Prima controlla se $P < 20$: in questo caso visualizza ``Apri il carico''. Altrimenti controlla
se $P > 90$: in questo caso visualizza ``Apri lo scarico''; in caso contrario visualizza ``Livello corretto''.}{%
In un negozio lo sconto dipende dalla spesa. Disegna il diagramma di flusso di un algoritmo che
calcola il totale da pagare.\\
\textbf{Dati da leggere:} $P$ = prezzo di un prodotto (in euro); $Q$ = quantità acquistata.\\
\textbf{Altre variabili:} $T$ = totale della spesa, cioè $T = P \cdot Q$.\\
Prima controlla se $T > 100$: in questo caso applica lo sconto del \qty{20}{\percent}
($T$ diventa $T \cdot 0{,}8$). Altrimenti controlla se $T > 50$: in questo caso applica lo sconto
del \qty{10}{\percent} ($T$ diventa $T \cdot 0{,}9$); in caso contrario non c'è sconto. Alla fine visualizza $T$.}

\begin{solution}
\begin{center}
\ABC{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini) at (0,0) {Inizio};
  \node[fcio]   (in)  at (0,-1.1) {Leggi $T_1$, $T_2$};
  \node[fcproc] (med) at (0,-2.2) {$T \gets (T_1+T_2)/2$};
  \node[fcdec]  (d1)  at (0,-3.55) {$T < 180$};
  \node[fcio]   (o1)  at (4.6,-3.55) {Scrivi ``Accendi la resistenza''};
  \node[fcdec]  (d2)  at (0,-5.25) {$T > 220$};
  \node[fcio]   (o2)  at (4.6,-5.25) {Scrivi ``Accendi la ventola''};
  \node[fcio]   (o3)  at (0,-6.75) {Scrivi ``Temperatura corretta''};
  \node[fcterm] (fin) at (0,-8.4) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (med);
  \draw[fcfreccia] (med) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,above] {Sì} (o1);
  \draw[fcfreccia] (d1) -- node[fcno,right] {No} (d2);
  \draw[fcfreccia] (d2) -- node[fcsi,above] {Sì} (o2);
  \draw[fcfreccia] (d2) -- node[fcno,right] {No} (o3);
  \draw[thick] (o1.east) -- (7.4,-3.55) |- (0,-7.65);
  \draw[thick] (o2.east) -- (7.4,-5.25);
  \draw[fcfreccia] (o3) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini) at (0,0) {Inizio};
  \node[fcio]   (in)  at (0,-1.1) {Leggi $L$, $H$};
  \node[fcproc] (med) at (0,-2.2) {$P \gets L/H \cdot 100$};
  \node[fcdec]  (d1)  at (0,-3.55) {$P < 20$};
  \node[fcio]   (o1)  at (4.4,-3.55) {Scrivi ``Apri il carico''};
  \node[fcdec]  (d2)  at (0,-5.25) {$P > 90$};
  \node[fcio]   (o2)  at (4.4,-5.25) {Scrivi ``Apri lo scarico''};
  \node[fcio]   (o3)  at (0,-6.75) {Scrivi ``Livello corretto''};
  \node[fcterm] (fin) at (0,-8.4) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (med);
  \draw[fcfreccia] (med) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,above] {Sì} (o1);
  \draw[fcfreccia] (d1) -- node[fcno,right] {No} (d2);
  \draw[fcfreccia] (d2) -- node[fcsi,above] {Sì} (o2);
  \draw[fcfreccia] (d2) -- node[fcno,right] {No} (o3);
  \draw[thick] (o1.east) -- (6.8,-3.55) |- (0,-7.65);
  \draw[thick] (o2.east) -- (6.8,-5.25);
  \draw[fcfreccia] (o3) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini) at (0,0) {Inizio};
  \node[fcio]   (in)  at (0,-1.1) {Leggi $P$, $Q$};
  \node[fcproc] (tot) at (0,-2.2) {$T \gets P \cdot Q$};
  \node[fcdec]  (d1)  at (0,-3.55) {$T > 100$};
  \node[fcproc] (s1)  at (3.8,-3.55) {$T \gets T \cdot 0{,}8$};
  \node[fcdec]  (d2)  at (0,-5.25) {$T > 50$};
  \node[fcproc] (s2)  at (3.8,-5.25) {$T \gets T \cdot 0{,}9$};
  \node[fcio]   (out) at (0,-7.15) {Scrivi $T$};
  \node[fcterm] (fin) at (0,-8.4) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (tot);
  \draw[fcfreccia] (tot) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,above] {Sì} (s1);
  \draw[fcfreccia] (d1) -- node[fcno,right] {No} (d2);
  \draw[fcfreccia] (d2) -- node[fcsi,above] {Sì} (s2);
  \draw[fcfreccia] (d2) -- node[fcno,right] {No} (out);
  \draw[thick] (s1.east) -- (5.4,-3.55) |- (0,-6.35);
  \draw[thick] (s2.east) -- (5.4,-5.25);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}
\end{center}
\ABC{%
  Il secondo controllo si fa solo se $T \geq 180$: quindi nel ramo ``No'' vale già $T \geq 180$ e
  basta controllare $T > 220$.}{%
  Il secondo controllo si fa solo se $P \geq 20$: quindi nel ramo ``No'' vale già $P \geq 20$ e
  basta controllare $P > 90$.}{%
  Il secondo controllo si fa solo se $T \leq 100$: quindi nel ramo ``No'' basta controllare $T > 50$.
  Sconto del \qty{20}{\percent}: si paga l'\qty{80}{\percent}, cioè $T \cdot 0{,}8$
  (oppure $T \gets T - T\cdot 20/100$).}
  \criterioABC
  {Inizio/Fine e simboli corretti (ovale, parallelogramma, rettangolo, rombo): 1 pt; lettura di $T_1$,
    $T_2$ e calcolo di $T$: 1 pt; decisione $T<180$ con i rami Sì/No: 2 pt; decisione $T>220$ nel ramo
    No: 1 pt; tre messaggi nei rami giusti e rami che si ricongiungono prima di Fine: 1 pt.}
  {Inizio/Fine e simboli corretti (ovale, parallelogramma, rettangolo, rombo): 1 pt; lettura di $L$, $H$
    e calcolo di $P$: 1 pt; decisione $P<20$ con i rami Sì/No: 2 pt; decisione $P>90$ nel ramo No:
    1 pt; tre messaggi nei rami giusti e rami che si ricongiungono prima di Fine: 1 pt.}
  {Inizio/Fine e simboli corretti (ovale, parallelogramma, rettangolo, rombo): 1 pt; lettura di $P$, $Q$
    e calcolo di $T$: 1 pt; decisione $T>100$ con sconto del 20\,\%: 2 pt; decisione $T>50$ nel ramo
    No con sconto del 10\,\%: 1 pt; rami che si ricongiungono e uscita di $T$: 1 pt.}
\end{solution}
\end{esercizio}
