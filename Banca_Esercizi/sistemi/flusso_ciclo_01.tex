% ---------------------------------------------------------------------
% Argomento: diagramma di flusso con ciclo (niente vettori: un solo dato
%            letto, il ciclo aggiorna variabili con nome dato nel testo)
%   fila A: serbatoio riempito a secchi -> numero di secchi e livello finale
%   fila B: magazzino svuotato a spedizioni -> numero di spedizioni e pezzi rimasti
%   (niente tempo: il ciclo conta operazioni, non minuti)
%   fila C: somma dei numeri da 1 a N e media
% Livello: intermedio   |   Tempo stimato: 15 min
% ---------------------------------------------------------------------
\begin{esercizio}[10]
Disegna il diagramma di flusso di un algoritmo che
\ABC{%
riempie un serbatoio vuoto versando secchi d'acqua. Legge la capacità $C$ del serbatoio e i litri
$P$ contenuti in un secchio. Finché il livello $L$ è minore di $C$ si versa un secchio: il livello
aumenta di $P$ e il contatore dei secchi $N$ aumenta di 1. Alla fine visualizza $N$ e $L$.}{%
svuota un magazzino con delle spedizioni. Legge il numero $Q$ di pezzi presenti in magazzino e il
numero $S$ di pezzi che partono con ogni spedizione. Finché $Q$ è maggiore o uguale a $S$ si fa una
spedizione: $Q$ diminuisce di $S$ e il contatore delle spedizioni $K$ aumenta di 1. Alla fine
visualizza $K$ e $Q$.}{%
legge un numero intero $N$ e calcola la somma $S = 1 + 2 + 3 + \ldots + N$ usando un contatore $i$
che parte da 1 e aumenta di 1 a ogni giro, finché $i$ non supera $N$. Dopo il ciclo calcola la
media $A = S/N$ e visualizza $S$ e $A$.}

\begin{solution}
\begin{center}
\ABC{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $C$, $P$};
  \node[fcinit] (ini0) at (0,-2.2) {$L \gets 0$, \ $N \gets 0$};
  \node[fcdec]  (d1)   at (0,-3.75) {$L < C$};
  \node[fcproc] (agg)  at (0,-5.3) {$L \gets L + P$};
  \node[fcproc] (inc)  at (0,-6.4) {$N \gets N + 1$};
  \node[fcio]   (out)  at (5.0,-3.75) {Scrivi $N$, $L$};
  \node[fcterm] (fin)  at (5.0,-4.95) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $Q$, $S$};
  \node[fcinit] (ini0) at (0,-2.2) {$K \gets 0$};
  \node[fcdec]  (d1)   at (0,-3.75) {$Q \geq S$};
  \node[fcproc] (agg)  at (0,-5.3) {$Q \gets Q - S$};
  \node[fcproc] (inc)  at (0,-6.4) {$K \gets K + 1$};
  \node[fcio]   (out)  at (5.0,-3.75) {Scrivi $K$, $Q$};
  \node[fcterm] (fin)  at (5.0,-4.95) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $N$};
  \node[fcinit] (ini0) at (0,-2.2) {$S \gets 0$, \ $i \gets 1$};
  \node[fcdec]  (d1)   at (0,-3.75) {$i \leq N$};
  \node[fcproc] (agg)  at (0,-5.3) {$S \gets S + i$};
  \node[fcproc] (inc)  at (0,-6.4) {$i \gets i + 1$};
  \node[fcproc] (med)  at (5.0,-3.75) {$A \gets S / N$};
  \node[fcio]   (out)  at (5.0,-4.95) {Scrivi $S$, $A$};
  \node[fcterm] (fin)  at (5.0,-6.1) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (med);
  \draw[fcfreccia] (med) -- (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}
\end{center}
\ABC{%
  Ogni giro del ciclo è un secchio versato: $N$ conta i giri. Il livello parte da 0 (serbatoio
  vuoto); quando il ciclo finisce, $L$ può superare di poco $C$ perché aumenta di $P$ alla volta.}{%
  Ogni giro del ciclo è una spedizione: $K$ conta i giri. Quando il ciclo finisce, in magazzino
  restano meno di $S$ pezzi (possono essere anche 0).}{%
  $S$ parte da 0 e $i$ da 1. Il ciclo si ripete finché $i \leq N$; la media si calcola una volta sola,
  dopo il ciclo.}
  \criterioABC
  {Inizio/Fine e simboli corretti (inizializzazione nell'esagono): 1 pt; lettura di $C$ e $P$: 1 pt; inizializzazione $L\gets 0$,
    $N\gets 0$: 2 pt; condizione del ciclo $L<C$ con rami Sì/No giusti: 2 pt; aggiornamenti
    $L\gets L+P$ e $N\gets N+1$ dentro il ciclo: 2 pt; ritorno alla condizione: 1 pt; uscita di $N$ e
    $L$ dopo il ciclo: 1 pt. Ciclo con condizione in fondo (ripeti\ldots finché $L \geq C$) corretto:
    stessi punti.}
  {Inizio/Fine e simboli corretti (inizializzazione nell'esagono): 1 pt; lettura di $Q$ e $S$: 1 pt; inizializzazione $K\gets 0$ (e
    nessuna inizializzazione di $Q$ dopo la lettura): 2 pt; condizione del ciclo $Q\geq S$ con rami
    Sì/No giusti: 2 pt; aggiornamenti $Q\gets Q-S$ e $K\gets K+1$ dentro il ciclo: 2 pt; ritorno alla
    condizione: 1 pt; uscita di $K$ e $Q$ dopo il ciclo: 1 pt. Ciclo con condizione in fondo
    corretto se gestisce il caso $Q<S$ iniziale: stessi punti.}
  {Inizio/Fine e simboli corretti (inizializzazione nell'esagono): 1 pt; lettura di $N$: 1 pt; inizializzazione $S\gets 0$, $i\gets 1$:
    2 pt; condizione del ciclo $i\leq N$ con rami Sì/No giusti: 2 pt; aggiornamenti $S\gets S+i$ e
    $i\gets i+1$ dentro il ciclo (in quest'ordine): 2 pt; ritorno alla condizione e media $S/N$ dopo il
    ciclo: 1 pt; uscita di $S$ e $A$: 1 pt. Ciclo con condizione in fondo corretto: stessi punti.}
\end{solution}
\end{esercizio}
