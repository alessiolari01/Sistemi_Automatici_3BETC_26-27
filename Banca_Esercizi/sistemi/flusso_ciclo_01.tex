% ---------------------------------------------------------------------
% Argomento: diagramma di flusso con ciclo (niente vettori: un solo dato
%            letto, il ciclo aggiorna variabili con nome dato nel testo)
%   fila A: serbatoio riempito da una pompa -> minuti e livello finale
%   fila B: forno che si raffredda -> minuti e temperatura finale
%   fila C: somma dei numeri da 1 a N e media
% Livello: intermedio   |   Tempo stimato: 15 min
% ---------------------------------------------------------------------
\begin{esercizio}[10]
Disegna il diagramma di flusso di un algoritmo che
\ABC{%
simula il riempimento di un serbatoio vuoto. Legge la capacità $C$ del serbatoio (in litri) e i
litri $P$ che la pompa immette ogni minuto. Finché il livello $L$ è minore di $C$, ogni minuto
il livello aumenta di $P$ e il contatore dei minuti $M$ aumenta di 1. Alla fine visualizza $M$
e $L$.}{%
simula il raffreddamento di un forno spento. Legge la temperatura iniziale $T$ (in °C) e i gradi
$D$ di cui la temperatura scende ogni minuto. Finché $T$ è maggiore o uguale a
\qty{50}{\degreeCelsius}, ogni minuto $T$ diminuisce di $D$ e il contatore dei minuti $M$ aumenta
di 1. Alla fine visualizza $M$ e $T$.}{%
legge un numero intero $N$ e calcola la somma $S = 1 + 2 + 3 + \ldots + N$ usando un contatore $i$
che parte da 1 e aumenta di 1 a ogni giro, finché $i$ non supera $N$. Dopo il ciclo calcola la
media $A = S/N$ e visualizza $S$ e $A$.}

\begin{solution}
\begin{center}
\ABC{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $C$, $P$};
  \node[fcproc] (ini0) at (0,-2.2) {$L \gets 0$, \ $M \gets 0$};
  \node[fcdec]  (d1)   at (0,-3.75) {$L < C$};
  \node[fcproc] (agg)  at (0,-5.3) {$L \gets L + P$};
  \node[fcproc] (inc)  at (0,-6.4) {$M \gets M + 1$};
  \node[fcio]   (out)  at (5.0,-3.75) {Scrivi $M$, $L$};
  \node[fcterm] (fin)  at (5.0,-4.95) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $T$, $D$};
  \node[fcproc] (ini0) at (0,-2.2) {$M \gets 0$};
  \node[fcdec]  (d1)   at (0,-3.75) {$T \geq 50$};
  \node[fcproc] (agg)  at (0,-5.3) {$T \gets T - D$};
  \node[fcproc] (inc)  at (0,-6.4) {$M \gets M + 1$};
  \node[fcio]   (out)  at (5.0,-3.75) {Scrivi $M$, $T$};
  \node[fcterm] (fin)  at (5.0,-4.95) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}{%
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[fcterm] (ini)  at (0,0) {Inizio};
  \node[fcio]   (in)   at (0,-1.1) {Leggi $N$};
  \node[fcproc] (ini0) at (0,-2.2) {$S \gets 0$, \ $i \gets 1$};
  \node[fcdec]  (d1)   at (0,-3.75) {$i \leq N$};
  \node[fcproc] (agg)  at (0,-5.3) {$S \gets S + i$};
  \node[fcproc] (inc)  at (0,-6.4) {$i \gets i + 1$};
  \node[fcproc] (med)  at (5.0,-3.75) {$A \gets S / N$};
  \node[fcio]   (out)  at (5.0,-4.95) {Scrivi $S$, $A$};
  \node[fcterm] (fin)  at (5.0,-6.1) {Fine};
  \draw[fcfreccia] (ini) -- (in);
  \draw[fcfreccia] (in) -- (ini0);
  \draw[fcfreccia] (ini0) -- (d1);
  \draw[fcfreccia] (d1) -- node[fcsi,right] {Sì} (agg);
  \draw[fcfreccia] (agg) -- (inc);
  \draw[fcfreccia] (inc.south) -- ++(0,-0.5) -| (-2.4,-2.95) -- (0,-2.95);
  \draw[fcfreccia] (d1) -- node[fcno,above] {No} (med);
  \draw[fcfreccia] (med) -- (out);
  \draw[fcfreccia] (out) -- (fin);
\end{tikzpicture}}
\end{center}
\ABC{%
  Il livello parte da 0 (serbatoio vuoto). Il ciclo si ripete finché $L < C$; quando esce, $L$ può
  superare di poco $C$ perché aumenta di $P$ alla volta.}{%
  Il ciclo si ripete finché $T \geq 50$; quando esce, $T$ è appena sotto i \qty{50}{\degreeCelsius}.}{%
  $S$ parte da 0 e $i$ da 1. Il ciclo si ripete finché $i \leq N$; la media si calcola una volta sola,
  dopo il ciclo.}
  \criterioABC
  {Inizio/Fine e simboli corretti: 1 pt; lettura di $C$ e $P$: 1 pt; inizializzazione $L\gets 0$,
    $M\gets 0$: 2 pt; condizione del ciclo $L<C$ con rami Sì/No giusti: 2 pt; aggiornamenti
    $L\gets L+P$ e $M\gets M+1$ dentro il ciclo: 2 pt; ritorno alla condizione: 1 pt; uscita di $M$ e
    $L$ dopo il ciclo: 1 pt. Ciclo con condizione in fondo (ripeti\ldots finché $L \geq C$) corretto:
    stessi punti.}
  {Inizio/Fine e simboli corretti: 1 pt; lettura di $T$ e $D$: 1 pt; inizializzazione $M\gets 0$ (e
    nessuna inizializzazione di $T$ dopo la lettura): 2 pt; condizione del ciclo $T\geq 50$ con rami
    Sì/No giusti: 2 pt; aggiornamenti $T\gets T-D$ e $M\gets M+1$ dentro il ciclo: 2 pt; ritorno alla
    condizione: 1 pt; uscita di $M$ e $T$ dopo il ciclo: 1 pt. Ciclo con condizione in fondo
    (ripeti\ldots finché $T<50$) corretto: stessi punti.}
  {Inizio/Fine e simboli corretti: 1 pt; lettura di $N$: 1 pt; inizializzazione $S\gets 0$, $i\gets 1$:
    2 pt; condizione del ciclo $i\leq N$ con rami Sì/No giusti: 2 pt; aggiornamenti $S\gets S+i$ e
    $i\gets i+1$ dentro il ciclo (in quest'ordine): 2 pt; ritorno alla condizione e media $S/N$ dopo il
    ciclo: 1 pt; uscita di $S$ e $A$: 1 pt. Ciclo con condizione in fondo corretto: stessi punti.}
\end{solution}
\end{esercizio}
